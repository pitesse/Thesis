\section{Implementation Pseudocode (Concise)}

\subsection{Batch OOF Loop}

\begin{verbatim}
for split in race_sequential_expanding_splits:
    fit calibration model on inner-train
    fit calibrator on inner-calibration scores
    lock threshold policy on calibration partition
    fit main XGBoost on outer-train
    score outer-test -> raw_proba, calibrated_proba
    append fold metrics + OOF rows
aggregate folds -> leaderboard + OOF artifact
\end{verbatim}

\subsection{MOA Prequential Vote Loop}

\begin{verbatim}
initialize OzaBoostAdwin learner
for each ARFF instance in order:
    votes = learner.getVotesForInstance(instance)
    positive_score = vote1 / (vote0 + vote1)
    emit row_index, y_true, y_pred, votes, positive_score
    learner.trainOnInstance(instance)
\end{verbatim}

\section{Major Engineering Corrections to Highlight}

\begin{itemize}
    \item Import-path hardening for package-safe execution of ML/MOA evaluators.
    \item MOA conversion from hard predicted labels to continuous vote-score artifacts.
    \item Explicit hard-decision frontier guard when score uniqueness $\leq 2$.
    \item Source-integrity audit for Batch probability diagnostics (Phase2B naming vs validated OOF source).
\end{itemize}
