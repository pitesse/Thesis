\section{Metrics and Ledger Semantics}

\subsection{Main metrics}
\begin{itemize}
    \item precision, successful-event coverage (recall-like), $F_{0.5}$;
    \item AP/PR for ranking diagnostics of score-based systems;
    \item support-aware diagnostics (Kappa, G-Mean, balanced accuracy; optional MCC where explicitly reported).
\end{itemize}

\subsection{Critical semantic distinction}
\begin{itemize}
    \item \textbf{strict operational precision}: used for apples-to-apples comparison;
    \item \textbf{matched-pit success rate}: diagnostic-only for Flink Strategy Engine strategy quality.
\end{itemize}

\section{Threats to Validity}

\begin{itemize}
    \item support sparsity (especially \texttt{pit\_success\_h2}) and race-level metric volatility;
    \item contract-policy coupling (threshold choice materially changes operating behavior);
    \item comparator assumptions and horizon strictness;
    \item surrogate explainability limitations for streaming models.
\end{itemize}
